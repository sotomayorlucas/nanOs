# Real ChaCha20-Poly1305 (RFC 8439) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the weak NERT crypto (ChaCha8 + fake 64-bit "Poly1305") with byte-exact RFC 8439 ChaCha20-Poly1305 AEAD, validated by RFC known-answer tests and QEMU interop.

**Architecture:** Layered crypto in the shared `nanOs/kernel/protocol/nert.c`: `chacha20_block` (20 rounds) → `chacha20_xor`/`chacha20_encrypt` (stream) → real `poly1305_mac`/`poly1305_verify` (130-bit, 16-byte tag) → `nert_aead_encrypt`/`nert_aead_decrypt` (full RFC 8439 seal/open with one-time Poly1305 key). The 3 wire call-sites call the AEAD; the epoch KDF uses the stream cipher. Clean wire break: `NERT_MAC_SIZE` 8→16.

**Tech Stack:** C99, freestanding -m32 kernel; tests are RFC 8439 KATs run in `nert_crypto_self_test()` at boot, observed on the QEMU serial log; then a 1-Queen/1-worker QEMU interop run.

## Global Constraints

- Byte-exact RFC 8439. AEAD: `otk = ChaCha20(key,nonce,ctr=0)[0:32]`; `ct = ChaCha20(key,nonce,ctr=1..) ^ pt`; `tag = Poly1305(otk, AAD‖pad16‖ct‖pad16‖le64(|AAD|)‖le64(|ct|))`, 16 bytes.
- `NERT_MAC_SIZE = 16` (was 8). Tag on the wire at `payload+packed_len`, 16 bytes.
- Nonce stays 12 bytes from `build_nonce(node_id, counter)`. AAD = the 20-byte NERT header.
- ChaCha20 = 20 rounds (10 double-rounds).
- Shared core compiled into BOTH: nanOs kernel (SRCS) and micrOs (`nert_shared.o` from `../nanOs/kernel/protocol/nert.c`). Every change to nert.c/nert.h must keep both building.
- nanOs builds with `-Werror -Wall -Wextra -Wno-unused-function -Wno-unused-but-set-variable`. micrOs has no -Werror.
- Test runner: build ISO, boot ONE nanOs node in QEMU (sandbox disabled), grep the serial log for the self-test lines. Interop uses the existing scratchpad scripts.
- `U8TO32LE(p) = p[0] | p[1]<<8 | p[2]<<16 | p[3]<<24` (little-endian load), already the convention in nert.c.

```c
void chacha20_xor(const uint8_t key[32], const uint8_t nonce[12], uint32_t counter,
                  const uint8_t *in, uint32_t len, uint8_t *out) {
    uint32_t key32[8], nonce32[3];
    uint8_t keystream[64];
    for (int i = 0; i < 8; i++)
        key32[i] = (uint32_t)key[i*4] | ((uint32_t)key[i*4+1]<<8) |
                   ((uint32_t)key[i*4+2]<<16) | ((uint32_t)key[i*4+3]<<24);
    for (int i = 0; i < 3; i++)
        nonce32[i] = (uint32_t)nonce[i*4] | ((uint32_t)nonce[i*4+1]<<8) |
                     ((uint32_t)nonce[i*4+2]<<16) | ((uint32_t)nonce[i*4+3]<<24);
    uint32_t off = 0;
    while (off < len) {
        chacha20_block(key32, nonce32, counter, keystream);
        counter++;
        uint32_t bl = (len - off > 64) ? 64 : (len - off);
        for (uint32_t i = 0; i < bl; i++) out[off+i] = in[off+i] ^ keystream[i];
        off += bl;
    }
    volatile uint8_t *vks = keystream; for (int i=0;i<64;i++) vks[i]=0;
    volatile uint32_t *vk = key32; for (int i=0;i<8;i++) vk[i]=0;
}
void chacha20_encrypt(const uint8_t key[32], const uint8_t nonce[12],
                      const uint8_t *pt, uint8_t len, uint8_t *ct) {
    chacha20_xor(key, nonce, 0, pt, len, ct);
}
```

Note `len` in `chacha20_xor` is `uint32_t` (AEAD data can exceed 255); `chacha20_encrypt` keeps `uint8_t len` for the existing callers.

```c
static int chacha20_self_test(void) {
    /* RFC 8439 2.3.2 keystream block */
    uint8_t key[32]; for (int i=0;i<32;i++) key[i]=i;
    uint8_t nonce[12]={0,0,0,0, 0,0,0,0x4a, 0,0,0,0};
    uint8_t blk_nonce[12]={0,0,0,9, 0,0,0,0x4a, 0,0,0,0};
    uint32_t k32[8],n32[3];
    for(int i=0;i<8;i++) k32[i]=U8TO32LE(&key[i*4]);
    for(int i=0;i<3;i++) n32[i]=U8TO32LE(&blk_nonce[i*4]);
    uint8_t block[64];
    chacha20_block(k32,n32,1,block);
    static const uint8_t exp_block[64] = { /* RFC 8439 2.3.2, 64 bytes, verbatim */ };
    if (memcmp(block, exp_block, 64) != 0) return -1;
    /* RFC 8439 2.4.2 ciphertext */
    static const uint8_t pt[114] = "Ladies and Gentlemen of the class of '99: If I could offer you only one tip for the future, sunscreen would be it.";
    static const uint8_t exp_ct[114] = { /* RFC 8439 2.4.2, 114 bytes, verbatim */ };
    uint8_t ct[114];
    chacha20_xor(key, nonce, 1, pt, 114, ct);
    if (memcmp(ct, exp_ct, 114) != 0) return -1;
    return 0;
}
```

```c
                           (p)[2]=(uint8_t)((v)>>16); (p)[3]=(uint8_t)((v)>>24); } while(0)

void poly1305_mac(const uint8_t otk[32], const uint8_t *m, uint32_t bytes, uint8_t mac[16]) {
    uint32_t r0,r1,r2,r3,r4, s1,s2,s3,s4;
    uint32_t h0=0,h1=0,h2=0,h3=0,h4=0, c;
    uint64_t d0,d1,d2,d3,d4;
    uint32_t g0,g1,g2,g3,g4; uint64_t f; uint32_t mask;

    r0 = (U8TO32LE(&otk[ 0])     ) & 0x3ffffff;
    r1 = (U8TO32LE(&otk[ 3]) >> 2) & 0x3ffff03;
    r2 = (U8TO32LE(&otk[ 6]) >> 4) & 0x3ffc0ff;
    r3 = (U8TO32LE(&otk[ 9]) >> 6) & 0x3f03fff;
    r4 = (U8TO32LE(&otk[12]) >> 8) & 0x00fffff;
    s1 = r1*5; s2 = r2*5; s3 = r3*5; s4 = r4*5;

    while (bytes >= 16) {
        h0 += (U8TO32LE(&m[ 0])     ) & 0x3ffffff;
        h1 += (U8TO32LE(&m[ 3]) >> 2) & 0x3ffffff;
        h2 += (U8TO32LE(&m[ 6]) >> 4) & 0x3ffffff;
        h3 += (U8TO32LE(&m[ 9]) >> 6) & 0x3ffffff;
        h4 += (U8TO32LE(&m[12]) >> 8) | (1u << 24);

        d1 += c; c = (uint32_t)(d1>>26); h1 = (uint32_t)d1 & 0x3ffffff;
        d2 += c; c = (uint32_t)(d2>>26); h2 = (uint32_t)d2 & 0x3ffffff;
        d3 += c; c = (uint32_t)(d3>>26); h3 = (uint32_t)d3 & 0x3ffffff;
        d4 += c; c = (uint32_t)(d4>>26); h4 = (uint32_t)d4 & 0x3ffffff;

        h2 += (U8TO32LE(&b[ 6]) >> 4) & 0x3ffffff;

        d2 += c; c = (uint32_t)(d2>>26); h2 = (uint32_t)d2 & 0x3ffffff;
        d3 += c; c = (uint32_t)(d3>>26); h3 = (uint32_t)d3 & 0x3ffffff;
        d4 += c; c = (uint32_t)(d4>>26); h4 = (uint32_t)d4 & 0x3ffffff;

    /* fully carry h */
    c = h1>>26; h1 &= 0x3ffffff; h2 += c; c = h2>>26; h2 &= 0x3ffffff;
    h3 += c; c = h3>>26; h3 &= 0x3ffffff; h4 += c; c = h4>>26; h4 &= 0x3ffffff;

    /* compute h - p */
    g0 = h0+5; c = g0>>26; g0 &= 0x3ffffff;
    g1 = h1+c; c = g1>>26; g1 &= 0x3ffffff;
    g2 = h2+c; c = g2>>26; g2 &= 0x3ffffff;
    g3 = h3+c; c = g3>>26; g3 &= 0x3ffffff;
    g4 = h4+c-(1u<<26);
    /* select h if h<p else h-p (constant time) */
    mask = (g4>>31)-1;

    mask = ~mask;
    h0 = (h0&mask)|g0; h1 = (h1&mask)|g1; h2 = (h2&mask)|g2; h3 = (h3&mask)|g3; h4 = (h4&mask)|g4;

    h0 = (h0    ) | (h1<<26);
    h1 = (h1>> 6) | (h2<<20);
    h2 = (h2>>12) | (h3<<14);
    h3 = (h3>>18) | (h4<< 8);
    /* mac = (h + s) % 2^128 */
    f = (uint64_t)h0 + U8TO32LE(&otk[16]);              h0=(uint32_t)f;
    f = (uint64_t)h1 + U8TO32LE(&otk[20]) + (f>>32);    h1=(uint32_t)f;
    f = (uint64_t)h2 + U8TO32LE(&otk[24]) + (f>>32);    h2=(uint32_t)f;
    f = (uint64_t)h3 + U8TO32LE(&otk[28]) + (f>>32);    h3=(uint32_t)f;
    U32TO8LE(&mac[ 0],h0); U32TO8LE(&mac[ 4],h1); U32TO8LE(&mac[ 8],h2); U32TO8LE(&mac[12],h3);
}
```

```c
int poly1305_verify(const uint8_t otk[32], const uint8_t *msg, uint32_t msg_len,
                    const uint8_t tag[16]) {
    uint8_t computed[16];
    poly1305_mac(otk, msg, msg_len, computed);
    uint8_t diff = 0;
    for (int i = 0; i < 16; i++) diff |= computed[i] ^ tag[i];
    return (diff == 0) ? 0 : -1;
}
```

```c
static void poly1305_key_gen(const uint8_t key[32], const uint8_t nonce[12], uint8_t otk[32]) {
    uint32_t k32[8], n32[3]; uint8_t blk[64];
    for (int i=0;i<8;i++) k32[i]=U8TO32LE(&key[i*4]);
    for (int i=0;i<3;i++) n32[i]=U8TO32LE(&nonce[i*4]);
    chacha20_block(k32, n32, 0, blk);         /* counter 0 */
    memcpy(otk, blk, 32);
    volatile uint8_t *v = blk; for (int i=0;i<64;i++) v[i]=0;
}
/* Poly1305 over AAD||pad16||CT||pad16||le64(aad_len)||le64(ct_len) */
static void aead_tag(const uint8_t otk[32], const uint8_t *aad, uint32_t aad_len,
                     const uint8_t *ct, uint32_t ct_len, uint8_t tag[16]) {
    uint8_t mac_data[16 + NERT_MAX_PAYLOAD + 16 + 16]; /* aad(<=16 header) + ct + pads + lens; sized to max */
    uint32_t o = 0;
    memcpy(mac_data+o, aad, aad_len); o += aad_len;
    while (o % 16) mac_data[o++] = 0;
    memcpy(mac_data+o, ct, ct_len); o += ct_len;
    while (o % 16) mac_data[o++] = 0;
    for (int i=0;i<8;i++) mac_data[o++] = (uint8_t)((uint64_t)aad_len >> (8*i));
    for (int i=0;i<8;i++) mac_data[o++] = (uint8_t)((uint64_t)ct_len  >> (8*i));
    poly1305_mac(otk, mac_data, o, tag);
}
void nert_aead_encrypt(const uint8_t key[32], const uint8_t nonce[12],
                       const uint8_t *aad, uint32_t aad_len,
                       const uint8_t *pt, uint32_t pt_len, uint8_t *ct, uint8_t tag[16]) {
    uint8_t otk[32];
    poly1305_key_gen(key, nonce, otk);
    chacha20_xor(key, nonce, 1, pt, pt_len, ct);   /* counter 1 */

}
```
